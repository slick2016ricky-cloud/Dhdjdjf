local monkey = script.Parent
local humanoid = monkey:FindFirstChildOfClass("Humanoid")

-- Speed boost
if humanoid then
	humanoid.WalkSpeed = 32 -- default is 16
end

-- RGB color cycle
local hue = 0
while task.wait(0.03) do
	hue = (hue + 0.01) % 1
	local color = Color3.fromHSV(hue, 1, 1)
	for _, part in ipairs(monkey:GetDescendants()) do
		if part:IsA("BasePart") then
			part.Color = color
		end
	end
end
